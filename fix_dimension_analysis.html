🔧 修复详细维度分析显示问题

此页面专门分析和修复index.html中详细维度分析文本缺失的问题

❌ 发现的问题

1. 元素ID不匹配问题

在HTML中，分析元素的ID格式是：

<p id="self-interest-analysis" class="text-sm text-gray-300"> <!-- 分析内容将通过JavaScript动态插入 --> </p>

但在JavaScript中查找时使用了错误的ID：

// 错误的查找方式 const analysisElement = document.getElementById(`${dimensionKey}-analysis`); // 这会生成: selfInterest-analysis (错误) // 实际需要的是: self-interest-analysis (正确)

2. 维度映射问题

JavaScript中的维度键名与HTML中的ID格式不一致：

// JavaScript中的维度键名 selfInterest, moralDisengagement, psychologicalEntitlement, selfCentered // HTML中的ID格式 self-interest, moral-disengagement, psychological-entitlement, self-centered

✅ 修复方案

修复后的updateDimensionUI函数

function updateDimensionUI(dimensionKey, dimensionName, elementIdPrefix = null) { console.log(`=== updateDimensionUI 调试 ${dimensionKey} ===`); const prefix = elementIdPrefix || dimensionKey; // 将驼峰式命名转换为连字符命名 const kebabCasePrefix = prefix.replace(/([a-z0-9])([A-Z])/g, '$1-$2').toLowerCase(); console.log(`维度: ${dimensionKey}, 前缀: ${prefix}, kebabCase: ${kebabCasePrefix}`); console.log(`分数: ${userScores[dimensionKey]}`); const scoreElement = document.getElementById(`${kebabCasePrefix}-score`); const barElement = document.getElementById(`${kebabCasePrefix}-bar`); // 修复：分析元素也使用连字符格式 const analysisElement = document.getElementById(`${kebabCasePrefix}-analysis`); console.log(`元素查找: score=${!!scoreElement}, bar=${!!barElement}, analysis=${!!analysisElement}`); console.log(`具体元素ID: ${kebabCasePrefix}-score, ${kebabCasePrefix}-bar, ${kebabCasePrefix}-analysis`); if (scoreElement) { const score = userScores[dimensionKey]; scoreElement.textContent = `${score}%`; console.log(`更新分数成功: ${dimensionKey} = ${score}%`); } else { console.log(`分数元素未找到: ${kebabCasePrefix}-score`); } if (barElement) { const score = userScores[dimensionKey]; barElement.style.width = `${score}%`; console.log(`更新进度条成功: ${dimensionKey} = ${score}%`); } else { console.log(`进度条元素未找到: ${kebabCasePrefix}-bar`); } if (analysisElement) { const score = userScores[dimensionKey]; const level = getLevel(score); analysisElement.textContent = analysisTexts[dimensionKey][level]; console.log(`更新分析文本成功: ${dimensionKey} - ${level}级`); } else { console.log(`分析元素未找到: ${kebabCasePrefix}-analysis`); } }
